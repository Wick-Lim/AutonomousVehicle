\begin{abstract}
Autoregressive vision-language-action (VLA) models \cite{brohan2023rt2,kim2024openvla,liu2023visual} decode tokens sequentially ($>300$\,ms/frame), exceeding the $50$\,ms budget of $20$\,Hz vehicle control \cite{dosovitskiy2017carla,li2022metadrive}. We implement a single-pass closed-loop driving stack coupling frozen DINOv2 (\texttt{facebook/dinov2-base}) \cite{oquab2024dinov2,caron2021emerging,dosovitskiy2021image} with frozen Laya (\texttt{convaiinnovations/laya}) \cite{laya2024nonautoregressive,warner2024modernbert,devlin2019bert} via a $4 \times 9$ spatial-temporal pooling projector ($1.85$M core parameters). Cropped $196 \times 392$ RGB frames map to 36 visual tokens with temporal motion residual $\Delta V_t = V_t - V_{t-1}$ and region-pooled auxiliary perception; one forward pass outputs maneuver (\texttt{choice}), speed tier (\texttt{score}), emergency brake (\texttt{noul}), and 6-waypoint ($3.0$\,s) trajectory residuals in $31.67$\,ms base ($32.85$\,ms full stack, $30.4$\,FPS on Ryzen~5 7500F / RTX~4070~Ti, FP32 $B=1$). Episode-block validation ($7,500$ frames, $M=15$ bins) temperature scaling \cite{guo2017calibration} reduces ECE from $0.0060$ to $<10^{-4}$ ($0.0142$ on unseen closed-loop vision-estimated test frames). Lateral waypoint step jumps ($3.5$\,m) and longitudinal speed-bucket hunting in CARLA 0.9.16 (\texttt{Town10HD\_Opt}) \cite{dupuis2010opendrive,hart1968formal} are eliminated using a 14-point ($28$\,m) boundary-zero-curvature spatial quintic spline $s(\tau) = 10\tau^3 - 15\tau^4 + 6\tau^5$ \cite{flash1985coordination,werling2010optimal}, decoupled chord-angle tracking \cite{thrun2006stanley,hoffmann2007autonomous,coulter1992implementation,paden2016survey}, kinematic bicycle curvature feedforward $\delta_{\text{FF}} = \arctan(L\kappa_{\text{eff}})$, 360$^\circ$ blind-spot interlock, and true constant-time-headway (CTH) ACC ($d^*(v_{\text{ego}}) = 8.0 + 1.2(v_{\text{ego}}/3.6)$, $k_g = 0.85$\,(km/h)/m) \cite{ioannou1993autonomous,rajamani2001experimental,treiber2000congested}. Across 50 unseen MetaDrive 0.4.3 episodes ($4,153$ frames), 10 paired scenario--weather urban episodes ($680$\,m, $1,981$ frames), and a 50-episode ($10 \times 5$ cross-weather, $3,400$\,m) CARLA 0.9.16 suite, the full system achieves $100.0\%$ ($10/10$) and $98.0\%$ ($49/50$) completion versus $60.0\%$ (keep-lane rule), $80.0\%$ (heuristic overtake FSM), and $70.0\%$ (300\,ms delayed VLA).
\end{abstract}

\section{Introduction and Problem Formulation}
Autoregressive decoders evaluate $P(y_{1:K} \mid I_t, \mathcal{T}_t) = \prod_{k=1}^{K} P(y_k \mid y_{<k}, I_t, \mathcal{T}_t)$ over $K \approx 15$ serial steps \cite{brohan2023rt2,kim2024openvla,liu2023visual}, whereas planning-oriented and sensor-fusion stacks \cite{hu2023planning,jiang2023vad,li2022bevformer,codevilla2018end,chen2020learning,prakash2021multi,chitta2022transfuser,jaeger2023hidden} require $\Delta t \le 50$\,ms. In our formulation, control at step $t$ reads three discrete decisions and $K_w=6$ ego-frame trajectory waypoints ($t_k = 0.5k$\,s) in one bidirectional pass \cite{devlin2019bert,warner2024modernbert,laya2024nonautoregressive}:
\begin{align}
    m_t &\in \{\texttt{keep\_lane}, \texttt{change\_left}, \nonumber\\
        &\quad\; \texttt{change\_right}, \texttt{follow\_lead}\}, \\
    s_t &\in \{0, 1, 2, 3\} \mapsto \{0, 20, 35, 50\}\,\text{km/h}, \\
    p_{\text{AEB}} &\in [0, 1], \quad
    \hat{\tau}_t = \{(\hat{x}_k, \hat{y}_k, \hat{v}_k)\}_{k=1}^{6}.
\end{align}

\section{Architecture, Multi-Task Training, and Calibration}
\label{sec:arch}
Camera frames are cropped ($\texttt{crop\_top}=0.25$, $\texttt{crop\_bottom}=0.10$) and resized to $196 \times 392$. Frozen DINOv2 ViT-B/14 ($p=14$) \cite{oquab2024dinov2,caron2021emerging,dosovitskiy2021image} extracts $F_{\text{patch}} \in \mathbb{R}^{B \times 392 \times 768}$ ($14 \times 28$ grid). \texttt{SpatialProjector} (\texttt{src/models/spatial\_projector.py}) pools $14 \times 28 \to 4 \times 9$ (36 tokens), adds $E_{\text{row}} + E_{\text{col}}$, projects $768 \to 1024$ ($1,851,392$ core parameters), and fuses temporal residual $\Delta V_t = V_t - V_{t-1}$ with region-pooled \texttt{AuxiliaryPerceptionHead} ($\hat{d}_{\text{lead}}, \hat{e}_y$, traffic light, hazard, blind spot):
\begin{align}
    G_t &= \text{AdaptiveAvgPool2d}_{(4,9)}(\tilde{F}_t) \in \mathbb{R}^{B \times 768 \times 4 \times 9}, \\
    V_t &= \text{MLP}\big(\text{Flatten}(G_t + E_{\text{row}} + E_{\text{col}})\big), \\
    \tilde{V}_t &= V_t + \tanh(\alpha_{\text{temp}})\, W_{\text{temp}}\big(V_t - V_{t-1}\big).
\end{align}
Multimodal sequence $Z = [e_{\text{[CLS]}}; \tilde{V}_t; E_{\text{telemetry}}; e_{\text{choice}}; e_{\text{score}}; e_{\text{noul}}]$ is evaluated in one pass $H = f_{\text{Laya}}(Z)$ \cite{laya2024nonautoregressive,warner2024modernbert}. The 6-waypoint trajectory head adds learned residuals $\Delta\hat{\tau} = W_{\text{traj}}\bar{H} \in \mathbb{R}^{6 \times 3}$ to kinematic prior $\tau^0_k$ ($t_k = 0.5k$\,s), supervised against ground-truth target $\tau^*_k = \big((v^*/3.6)t_k,\, y_{\text{shift}}(m^*)s(k/6),\, v^*\big)$:
\begin{align}
    \hat{\tau}_{1:6} &= \tau^0_{1:6} + \text{diag}(1.5, 0.35, 3.0)\, W_{\text{traj}}\bar{H}, \\
    \mathcal{L} &= w_m \mathcal{L}_{\text{CE}}^{(m)} + w_s \mathcal{L}_{\text{CE}}^{(s)} + w_a \mathcal{L}_{\text{BCE}}^{(b)} \nonumber\\
                &\quad + w_\tau \mathcal{L}_{\text{SmoothL1}}(\hat{\tau}_{1:6}, \tau^*_{1:6}) + w_p \mathcal{L}_{\text{aux}}.
\end{align}

Stage~1 trains \texttt{SpatialProjector} and \texttt{trajectory\_head} for 15 epochs (batch 32, AdamW \cite{loshchilov2019decoupled} $\text{lr}=3\times 10^{-4}$, $(w_m,w_s,w_a,w_\tau,w_p)=(1.0,1.0,2.0,0.5,0.2)$, $3\times$/$4\times$ rare-event oversampling). To prevent adjacent-frame leakage, $50,000$ frames ($500$ episodes) are partitioned by contiguous episode blocks (\texttt{split\_by\_episode=True}: episodes $0..424$, $42,500$ frames for training; episodes $425..499$, $7,500$ frames for validation). Stage~2 (\texttt{03\_calibrate.py}) fits $(T_m, T_s, T_b) = (0.050, 0.050, 0.100)$ via L-BFGS \cite{guo2017calibration} ($M=15$ bins), reducing validation ECE from $0.0060$ to $<10^{-4}$; on independent closed-loop test episodes (seeds $5000+$), ECE is $0.0000$ against deterministic \texttt{ExpertLabeler} thresholds and $0.0142$ under vision-estimated perception inputs.

\section{Safety Shield, Spatial Quintic Spline, Curvature Feedforward, and CTH-ACC}
\label{sec:control}
We evaluate two perception modes (\texttt{--perception-mode gt} vs.\ \texttt{vision\_estimated} using \texttt{AuxiliaryPerceptionHead} outputs $\hat{d}_{\text{lead}}, \hat{e}_y, \hat{c}_{\text{TL}}, \hat{c}_{\text{haz}}, \hat{p}_{\text{BS}}$) and two shield modes (\texttt{--shield-mode on} vs.\ \texttt{off}, where \texttt{off} triggers braking solely when Laya $\hat{p}_{\text{AEB}} \ge 0.70$). When enabled, Rule~1 forces $u_{\text{tb}} = -1.0$ if $\hat{p}_{\text{AEB}} \ge 0.70$, $\texttt{traffic\_light}=\texttt{red}$, pedestrian $d \le 16.0$\,m, or $d_{\text{lead}} \le 11.5$\,m; Rule~2 reverts lane changes to \texttt{keep\_lane} if $\max_c \hat{p}^{(m)}_c < 0.80$, if $\texttt{adj\_wp.lane\_id} \times \texttt{wp.lane\_id} \le 0$ \cite{dupuis2010opendrive}, or if 360$^\circ$ blind-spot box $x \in [-10.0, 7.5]$\,m, $|y| \in [1.8, 5.0]$\,m is occupied.

For lane transitions over $N_b = 14$ waypoints ($28.0$\,m, $\tau_k = k/(N_b - 1)$), 3D coordinates are blended via a spatial quintic polynomial enforcing zero boundary slope and curvature $s'(0)=s'(1)=s''(0)=s''(1)=0$ (equivalent to minimum jerk under constant speed) \cite{flash1985coordination,werling2010optimal}:
\begin{align}
    s(\tau) &= 10\tau^3 - 15\tau^4 + 6\tau^5, \\
    P^{\text{blend}}_k &= \big(1 - s(\tau_k)\big) P^{\text{orig}}_k + s(\tau_k) P^{\text{adj}}_k.
\end{align}
Continuous projection onto $\overline{P_i P_{i+1}}$ gives $P_{\text{proj}}$ and lookahead $P_{\text{look}}$ at $L_{\text{look}} = \text{clip}(7.5 + 0.18 v_{\text{kmh}}, 7.5, 13.5)$\,m \cite{thrun2006stanley,hoffmann2007autonomous,coulter1992implementation,paden2016survey}. Predicted waypoint $k=3$ ($t=1.5$\,s, $(\hat{x}_3, \hat{y}_3)$) is fused with map curvature $\kappa_{\text{map}}$ into a kinematic bicycle curvature feedforward term $\delta_{\text{FF}} = \arctan(L \kappa_{\text{eff}})$ ($L=2.85$\,m):
\begin{align}
    \kappa_{\text{traj}} &= \frac{2(\hat{y}_3 - e_y)}{\max(3.5, \hat{x}_3)^2 + (\hat{y}_3 - e_y)^2}, \\
    \kappa_{\text{eff}} &= 0.70(-0.045\kappa_{\text{map}}) + 0.30\kappa_{\text{traj}}, \\
    \delta_{\text{raw}} &= \text{clip}\big(\delta_{\text{PID}}(e_y, e_\psi) + 0.12\,\delta_{\text{FF}}/0.70, -1, 1\big), \\
    \Delta \delta_{\max} &= \frac{1.1\Delta t}{1 + 0.012 v_{\text{kmh}}}, \\
    \delta_t &= \delta_{t-1} + \text{clip}\!\big(\alpha(\delta_{\text{raw}} - \delta_{t-1}), -\Delta\delta_{\max}, \Delta\delta_{\max}\big).
\end{align}
True constant-time-headway (CTH) ACC ($d_{\text{lead}} < 38$\,m, $v_{\text{lead}} \ge 8$\,km/h, $d_0 = 8.0$\,m, $\tau_h = 1.2$\,s, $k_g = 0.85$\,(km/h)/m $\equiv 0.236\,\text{s}^{-1}$) \cite{ioannou1993autonomous,rajamani2001experimental,treiber2000congested} follows:
\begin{align}
    d^*(v_{\text{ego,kmh}}) &= d_0 + \tau_h \left(\frac{v_{\text{ego,kmh}}}{3.6}\right), \\
    v_{\text{raw}} &= v_{\text{lead}} + k_g \big(d_{\text{lead}} - d^*(v_{\text{ego,kmh}})\big), \\
    v_{\text{ACC}} &= \text{clip}\!\big(v_{\text{raw}},\, 0.0,\, \min(35.0,\, v_{\text{lead}} + 6.0)\big).
\end{align}

\begin{verbatim}
# src/control/shield_pid.py

traj_curv = (2.0 * ly) / (lx * lx + ly * ly)

\end{verbatim}
\end{scriptsize}
